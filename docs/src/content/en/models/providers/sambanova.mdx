const agent = new Agent({

  name: "Dynamic Agent",
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === "complex";
    return useAdvanced
      ? "sambanova/Llama-4-Maverick-17B-128E-Instruct"
      : "sambanova/Meta-Llama-3.3-70B-Instruct";
  }
});
