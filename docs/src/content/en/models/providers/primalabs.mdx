```typescript title="src/mastra/agents/my-agent.ts"
import { Agent } from "@mastra/core/agent";

const agent = new Agent({
  id: "custom-agent",
  name: "custom-agent",
  model: {
    url: "https://api.primalabs.ai/v1",
    id: "primalabs/primalabs-ai/DeepSeek-V4.1-Flash-1017",
    apiKey: process.env.PRIMALABS_API_KEY,

      "X-Custom-Header": "value"

});
```

### Dynamic model selection

```typescript title="src/mastra/agents/my-agent.ts"
import { Agent } from "@mastra/core/agent";

const agent = new Agent({
  id: "dynamic-agent",
  name: "Dynamic Agent",
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === "complex";
    return useAdvanced
      ? "primalabs/primalabs-ai/Nemotron-3.5-Super-VL-NVFP4"
      : "primalabs/primalabs-ai/DeepSeek-V4.1-Flash-1017";
  }
});
```
