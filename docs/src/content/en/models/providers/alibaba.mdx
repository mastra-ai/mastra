```typescript title="src/mastra/agents/my-agent.ts"
  id: "custom-agent",
  name: "custom-agent",
  model: {
```

### Dynamic model selection

```typescript title="src/mastra/agents/my-agent.ts"
const agent = new Agent({
  id: "dynamic-agent",
  name: "Dynamic Agent",
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === "complex";
    return useAdvanced
      ? "alibaba/qwq-plus"
      : "alibaba/deepseek-v4-flash-0731";
  }
});
```
