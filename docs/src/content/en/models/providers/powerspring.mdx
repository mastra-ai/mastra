## Advanced configuration

### Custom headers

```typescript title="src/mastra/agents/my-agent.ts"
import { Agent } from "@mastra/core/agent";

const agent = new Agent({
  id: "custom-agent",
  name: "custom-agent",
  model: {
    url: "https://api.powerspring.eu/v1",
    id: "powerspring/deepseek-v4.1-flash",
    apiKey: process.env.POWERSPRING_API_KEY,
    headers: {
      "X-Custom-Header": "value"
    }
  }
});
```

### Dynamic model selection

```typescript title="src/mastra/agents/my-agent.ts"
import { Agent } from "@mastra/core/agent";

const agent = new Agent({
  id: "dynamic-agent",
  name: "Dynamic Agent",
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === "complex";
    return useAdvanced
      ? "powerspring/qwen3.8-27b"
      : "powerspring/deepseek-v4.1-flash";
  }
});
```
