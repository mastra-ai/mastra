## Advanced configuration

### Custom headers

```typescript title="src/mastra/agents/my-agent.ts"
import { Agent } from "@mastra/core/agent";

const agent = new Agent({
  id: "custom-agent",
  name: "custom-agent",
  model: {
    url: "https://tokenhub.tencentmaas.com/plan/v3",
    id: "tencent-token-plan/deepseek/deepseek-flash",
    apiKey: process.env.TENCENT_TOKEN_PLAN_API_KEY,
    headers: {
      "X-Custom-Header": "value"
    }
  }
});
```

### Dynamic model selection

```typescript title="src/mastra/agents/my-agent.ts"
import { Agent } from "@mastra/core/agent";

const agent = new Agent({
  id: "dynamic-agent",
  name: "Dynamic Agent",
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === "complex";
    return useAdvanced
      ? "tencent-token-plan/minimax-m3"
      : "tencent-token-plan/deepseek/deepseek-flash";
  }
});
```
