## Advanced configuration

### Custom headers

```typescript title="src/mastra/agents/my-agent.ts"
import { Agent } from "@mastra/core/agent";

const agent = new Agent({
  id: "custom-agent",
  name: "custom-agent",
  model: {
    url: "https://model.graphn.ai/v1",
    id: "graphn/deepseek-v4.1-flash",
    apiKey: process.env.GRAPHN_API_KEY,
    headers: {
      "X-Custom-Header": "value"
    }
  }
});
```

### Dynamic model selection

```typescript title="src/mastra/agents/my-agent.ts"
import { Agent } from "@mastra/core/agent";

const agent = new Agent({
  id: "dynamic-agent",
  name: "Dynamic Agent",
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === "complex";
    return useAdvanced
      ? "graphn/qwen3.8-27b"
      : "graphn/deepseek-v4.1-flash";
  }
});
```
